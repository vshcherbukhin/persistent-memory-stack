# =============================================================================
# persistent-memory-graphiti — Python deps (Phase 4)
#
# Internal-only temporal-graph microservice (FastAPI), called ONLY by the TS
# API/worker over the private compose network. It does NO auth — it trusts the
# group_ids the API passes (the API is the choke-point that derives readableTeams
# from identity). See plan §"Access model & security" + per-phase note P4.
#
# graphiti-core API surface (reconfirmed against PyPI + GitHub @ v0.29.2):
#   - requires-python ">=3.10,<4"  → python:3.12-slim is in range.
#   - BASE deps already pull in `neo4j>=5.26.0` AND `openai>=1.91.0`, so:
#       * the GRAPH_BACKEND=neo4j swap needs NO extra (Neo4jDriver ships in core).
#       * the Ollama embedder/LLM path is the OpenAI-compatible client + base_url
#         (http://host:11434/v1) — covered by the base `openai` dep, no extra.
#   - extras pinned below:
#       * [falkordb]  → falkordb>=1.1.2,<2.0.0  (PRIMARY graph backend driver;
#                       1.1.2 is the ENGINE/module min, NOT the Docker image tag —
#                       the falkordb/falkordb:v4.x image far exceeds it).
#       * [anthropic] → anthropic>=0.49.0       (default cloud extraction LLM:
#                       EXTRACTION_PROVIDER=anthropic, claude-haiku-4-5).
#   - `voyageai` added explicitly so the Voyage embedder path is first-class
#     (graphiti-core exposes VoyageAIEmbedder; the dep is NOT in the falkordb/
#     anthropic extras). Cloud OpenAI embeddings ride the base `openai` dep.
#   - SEMAPHORE_LIMIT is read from the environment by graphiti-core directly
#     (caps concurrent LLM/embedder ops to dodge 429s); no package needed.
#
# Pins are exact (==) for reproducible images. Bump deliberately, not by float.
# =============================================================================

# --- Graphiti core + graph backend + extraction LLM extras -------------------
graphiti-core[falkordb,anthropic]==0.29.2
# Graphiti 0.29.2 still sends sampling kwargs that Anthropic SDK 1.x removed
# from messages.create(). Pin the tested SDK while the service compatibility
# adapter normalizes that upstream call at runtime.
anthropic==1.3.0
# FalkorDB currently declares only redis>=7.1.0. Keep this known-compatible
# pair explicit: redis-py 8.1.0 passes an async-only internal pool option to
# FalkorDB's synchronous cluster probe and prevents Graphiti from starting.
falkordb==1.6.2
redis==8.0.1

# --- Embedder providers (server-side graph-node embeddings) ------------------
# IMPORTANT: this embedder embeds GRAPHITI'S OWN graph nodes/edges and is ALWAYS
# server-side — it is unrelated to the Qdrant CHUNK-vector Mode A/B client-bridge
# toggle (that lives in P5/the TS side). In Mode B the recommended Graphiti
# embedder is cloud (Voyage/OpenAI) or a small server Ollama; graph nodes are few.
#   * openai   → also Graphiti's base dep, but pinned here so the Ollama
#                OpenAI-compatible embedder/LLM path and cloud OpenAI embeddings
#                are reproducible.
#   * voyageai → Voyage embedder path (EMBED_PROVIDER=voyage), opt-in cloud.
openai==1.99.9
voyageai==0.3.5

# --- Web framework / ASGI server ---------------------------------------------
# uvicorn[standard] pulls uvloop + httptools + websockets for prod-grade serving.
fastapi==0.115.6
uvicorn[standard]==0.34.0

# --- Config / validation -----------------------------------------------------
# pydantic is already a graphiti-core base dep (>=2.11.5); pinned here for the
# service's own request/response models. pydantic-settings reads config from env.
pydantic==2.11.5
pydantic-settings==2.7.1
